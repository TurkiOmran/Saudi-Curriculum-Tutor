if month&lt;1 or month&gt;13:
print("رقم الشهر خاطئ")

if month&gt;=9 and month&lt;=11:
print("فصل الخريف")
elif month==12 or month&gt;=1 and month&lt;=2:
print("فصل الشتاء")
elif month&gt;=3 and month&lt;=5:
print("فصل الربيع")
else:
print("فصل الصيف")
