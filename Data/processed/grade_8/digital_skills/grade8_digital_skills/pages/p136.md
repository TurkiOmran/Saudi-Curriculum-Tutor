```python
print("من فضلك أدخل الدرجة:")
g=int(input())
if g&lt;0 or g&gt;10:
print("درجة غير صالحة")
elif g&gt;=8:
print("ممتاز")
elif g&gt;=5:
print("جيد جدًا")
else:
print("اجتهد أكثر")
```
